# --- Simple Local Python HTTP Server ---
import http.server
import socketserver

PORT = 8000
Handler = http.server.SimpleHTTPRequestHandler

with socketserver.TCPServer(("", PORT), Handler) as httpd:
    print("Serving on port:", PORT)
    httpd.serve_forever()
