// --- JavaScript Terminal Logic ---
function handleCliCommand(event) {
    if (event.key === 'Enter') {
        const cmd = event.target.value.toLowerCase().trim();
        if (cmd === 'help') {
            console.log("Commands: help, clear, time");
        }
    }
}
